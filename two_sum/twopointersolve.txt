import java.util.Arrays;

class Solution {
    public int[] twoSum(int[] nums, int target) {
        int n = nums.length;
        int[][] pairs = new int[n][2];

        // store number and its original index
        for (int i = 0; i < n; i++) {
            pairs[i][0] = nums[i];
            pairs[i][1] = i;
        }

        // sort by number value
        Arrays.sort(pairs, (a, b) -> Integer.compare(a[0], b[0]));

        int left = 0;
        int right = n - 1;

        // two-pointer search
        while (left < right) {
            int sum = pairs[left][0] + pairs[right][0];

            if (sum == target) {
                // return original indices
                return new int[] { pairs[left][1], pairs[right][1] };
            } else if (sum < target) {
                left++;
            } else {
                right--;
            }
        }

        // should never happen (as per problem statement)
        return new int[0];
    }
}
